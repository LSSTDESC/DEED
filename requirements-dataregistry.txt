# Install with:
#   pip install -r requirements.txt
#   pip install --no-deps -r requirements-dataregistry.txt
#
# --no-deps is required: lsstdesc-dataregistry declares `psycopg2`, which ships
# no Linux wheels, so pip would try to build it from source and fail on
# Perlmutter for lack of pg_config. requirements.txt installs psycopg2-binary
# instead, which provides the same importable `psycopg2` module as a prebuilt
# wheel, along with the rest of dataregistry's dependencies.
lsstdesc-dataregistry==1.7.0
